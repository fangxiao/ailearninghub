进阶 快速启动模板

为了方便大家快速上手，这里提供一个完整的启动脚本：

bash
# 一键部署脚本 # 1. 安装Ollama curl -fsSL https://ollama.com/install.sh | sh # 2. 拉取模型 ollama pull llama3:8b # 3. 创建项目目录 mkdir local-kb && cd local-kb # 4. 安装依赖 pip install langchain chromadb sentence-transformers gradio pypdf # 5. 创建应用 cat > app.py << 'EOF' import gradio as gr from langchain.vectorstores import Chroma from langchain.embeddings import HuggingFaceEmbeddings from langchain.chains import RetrievalQA from langchain.llms import Ollama embeddings = HuggingFaceEmbeddings(model_name="all-MiniLM-L6-v2") vector_store = Chroma(persist_directory="./chroma_db", embedding_function=embeddings) llm = Ollama(model="llama3:8b") retriever = vector_store.as_retriever(search_kwargs={"k": 3}) qa_chain = RetrievalQA.from_chain_type(llm=llm, chain_type="stuff", retriever=retriever) def chat(message, history): result = qa_chain({"query": message}) return result["result"] with gr.Blocks(title="本地知识库") as demo: gr.Markdown("# 📚 本地知识库问答系统") gr.ChatInterface(chat, title="知识库助手") if __name__ == "__main__": demo.launch(share=False) EOF # 6. 启动服务 python app.py
