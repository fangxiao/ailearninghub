import gradio as gr import json class EnglishLearningPlatform: def __init__(self): self.vocabulary_reviewer = None self.grammar_learner = None self.reading_learner = None self.speaking_practice = None self.writing_coach = None self._init_learners() def _init_learners(self): try: from vocabulary_reviewer import VocabularyReviewer self.vocabulary_reviewer = VocabularyReviewer() except ImportError: print("Vocabulary module not available") try: from grammar_learner import GrammarLearner self.grammar_learner = GrammarLearner() except ImportError: print("Grammar module not available") try: from reading_learner import ReadingComprehensionLearner self.reading_learner = ReadingComprehensionLearner() except ImportError: print("Reading module not available") try: from speaking_practice import SpeakingPractice self.speaking_practice = SpeakingPractice() except ImportError: print("Speaking module not available") try: from writing_coach import WritingCoach self.writing_coach = WritingCoach() except ImportError: print("Writing module not available") def build_interface(self): with gr.Blocks(title="英语学习Agent", theme=gr.themes.Soft()) as demo: gr.Markdown( """ # 🎓 英语学习Agent 一个基于本地大模型的智能英语学习平台，支持词汇、语法、阅读、口语、写作全方位训练。 """ ) with gr.Tabs(): with gr.TabItem("📚 词汇学习"): self._build_vocabulary_tab() with gr.TabItem("📖 语法学习"): self._build_grammar_tab() with gr.TabItem("📝 阅读理解"): self._build_reading_tab() with gr.TabItem("🗣️ 口语练习"): self._build_speaking_tab() with gr.TabItem("✍️ 写作辅助"): self._build_writing_tab() return demo
python
def _build_vocabulary_tab(self): if not self.vocabulary_reviewer: gr.Markdown("词汇模块未加载") return word_input = gr.Textbox(label="输入单词") add_btn = gr.Button("添加单词") result_output = gr.JSON(label="单词卡") start_review_btn = gr.Button("开始复习") review_output = gr.JSON(label="复习内容") add_btn.click( fn=self.vocabulary_reviewer.add_new_word, inputs=[word_input], outputs=[result_output] ) start_review_btn.click( fn=self.vocabulary_reviewer.start_review, outputs=[review_output] ) def _build_grammar_tab(self): if not self.grammar_learner: gr.Markdown("语法模块未加载") return topic_dropdown = gr.Dropdown( choices=["基础句型", "名词", "冠词", "动词", "一般现在时", "一般过去时", "情态动词"], label="选择语法点" ) start_btn = gr.Button("开始学习") explanation_output = gr.Markdown(label="讲解") exercises_output = gr.JSON(label="练习题") start_btn.click( fn=self._format_grammar_result, inputs=[topic_dropdown], outputs=[explanation_output, exercises_output] ) def _format_grammar_result(self, topic): result = self.grammar_learner.start_topic(topic) if result["status"] == "error": return result["message"], [] explanation = result["explanation"] md_text = f"## {explanation.get('topic', '')}\n\n" md_text += f"**定义**：{explanation.get('definition', '')}\n\n" md_text += f"**规则**：\n" for rule in explanation.get("rules", []): md_text += f"- {rule}\n" return md_text, result["exercises"]
python
def _build_reading_tab(self): if not self.reading_learner: gr.Markdown("阅读模块未加载") return get_article_btn = gr.Button("获取推荐文章") article_output = gr.Markdown(label="文章内容") summary_output = gr.JSON(label="摘要") start_quiz_btn = gr.Button("开始测验") quiz_output = gr.JSON(label="题目") get_article_btn.click( fn=self._format_article_result, outputs=[article_output, summary_output] ) start_quiz_btn.click( fn=self.reading_learner.start_quiz, outputs=[quiz_output] ) def _format_article_result(self): result = self.reading_learner.get_recommended_article() if"error" in result: return result["error"], [] article = result["article"] md_text = f"## {article.get('title', '')}\n\n" md_text += f"**难度**：{article.get('cefr_level', '')}\n\n" md_text += article.get("content", "") return md_text, result["summary"]
python
def _build_speaking_tab(self): if not self.speaking_practice: gr.Markdown("口语模块未加载") return scenario_dropdown = gr.Dropdown( choices=["GREETING", "ORDERING_FOOD", "SHOPPING", "TRAVEL", "JOB_INTERVIEW"], label="选择场景" ) start_btn = gr.Button("开始对话") opening_output = gr.Markdown(label="对话开始") user_input = gr.Textbox(label="你的回答") send_btn = gr.Button("发送") reply_output = gr.Markdown(label="回复") evaluation_output = gr.JSON(label="评估") start_btn.click( fn=self._format_speaking_start, inputs=[scenario_dropdown], outputs=[opening_output] ) send_btn.click( fn=self._format_speaking_reply, inputs=[user_input], outputs=[reply_output, evaluation_output] ) def _format_speaking_start(self, scenario): result = self.speaking_practice.start_practice(scenario) returnf"## {result.get('scenario', '')}\n\n**AI**: {result.get('opening', '')}" def _format_speaking_reply(self, input_text): result = self.speaking_practice.speak(input_text) returnf"**AI**: {result.get('reply', '')}", result["evaluation"]
python
def _build_writing_tab(self): if not self.writing_coach: gr.Markdown("写作模块未加载") return category_input = gr.Textbox(label="主题类别（如：教育、科技、生活）") suggest_topics_btn = gr.Button("推荐题目") topics_output = gr.JSON(label="题目列表") topic_input = gr.Textbox(label="输入写作题目") get_outline_btn = gr.Button("获取大纲") outline_output = gr.JSON(label="大纲") essay_input = gr.Textbox(label="输入作文", lines=10) evaluate_btn = gr.Button("评估作文") evaluation_output = gr.JSON(label="评估结果") suggest_topics_btn.click( fn=self.writing_coach.suggest_topics, inputs=[category_input], outputs=[topics_output] ) get_outline_btn.click( fn=self.writing_coach.get_outline, inputs=[topic_input], outputs=[outline_output] ) evaluate_btn.click( fn=self.writing_coach.evaluate_essay, inputs=[essay_input], outputs=[evaluation_output] ) # 运行平台 if __name__ == "__main__": platform = EnglishLearningPlatform() demo = platform.build_interface() demo.launch(server_name="0.0.0.0", server_port=7860)

⚠️ 常见问题

1. 模块导入失败：确保所有模块文件在同一目录下

2. Gradio界面样式问题：尝试更换theme参数

3. 界面响应慢：优化Ollama调用速度（参考【本地大模型知识库系列】第10篇）

4. 端口被占用：修改server_port参数
