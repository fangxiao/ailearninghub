自适应难度阅读

英语学习Agent系列第04篇 | 阅读理解

阅读理解是英语学习的核心能力之一。传统阅读材料往往难度固定，无法根据学习者水平动态调整。我们将构建一个自适应难度的阅读理解系统，根据学习者的表现自动调整文章难度，并生成针对性的练习题目。

01 文章难度评估与分级

使用Flesch-Kincaid可读性指数和词汇难度来评估文章难度：

python
import textstat import re class DifficultyEvaluator: def __init__(self): self.cefr_levels = { 1: "A1", 2: "A2", 3: "B1", 4: "B2", 5: "C1", 6: "C2" } def evaluate(self, text): fk_grade = textstat.flesch_kincaid_grade(text) flesch_ease = textstat.flesch_reading_ease(text) sentence_count = textstat.sentence_count(text) word_count = textstat.lexicon_count(text, removepunct=True) avg_sentence_length = word_count / sentence_count if sentence_count > 0 else0 cefr_level = self._map_to_cefr(fk_grade) return{ "fk_grade": round(fk_grade, 2), "flesch_ease": round(flesch_ease, 2), "avg_sentence_length": round(avg_sentence_length, 2), "word_count": word_count, "sentence_count": sentence_count, "cefr_level": cefr_level, "difficulty_score": self._calculate_difficulty_score(fk_grade, avg_sentence_length) } def _map_to_cefr(self, fk_grade): if fk_grade <= 3: return"A1" elif fk_grade <= 5: return"A2" elif fk_grade <= 7: return"B1" elif fk_grade <= 9: return"B2" elif fk_grade <= 11: return"C1" else: return"C2" def _calculate_difficulty_score(self, fk_grade, avg_sentence_length): return round(fk_grade * 0.6 + avg_sentence_length * 0.4, 2)

02 文章库管理与检索

使用向量数据库存储文章，支持按难度检索（关于向量数据库的使用，请参考【本地大模型知识库系列】第03篇）：

python
import chromadb import json class ArticleLibrary: def __init__(self, persist_directory="./chroma_db"): self.client = chromadb.PersistentClient(path=persist_directory) self.collection = self.client.get_or_create_collection("english_articles") self.evaluator = DifficultyEvaluator() def add_article(self, title, content, category="general"): difficulty = self.evaluator.evaluate(content) metadata = { "title": title, "category": category, "cefr_level": difficulty["cefr_level"], "difficulty_score": difficulty["difficulty_score"], "word_count": difficulty["word_count"] } self.collection.add( documents=[content], metadatas=[metadata], ids=[title] ) return difficulty def get_articles_by_difficulty(self, cefr_level, limit=5): results = self.collection.query( query_texts=[" "], where={"cefr_level": cefr_level}, n_results=limit ) return[{ "title": meta["title"], "content": doc, "cefr_level": meta["cefr_level"], "difficulty_score": meta["difficulty_score"] } for doc, meta in zip(results["documents"][0], results["metadatas"][0])] def search_articles(self, query, cefr_level=None, limit=5): where_clause = {"cefr_level": cefr_level} if cefr_level elseNone results = self.collection.query( query_texts=[query], where=where_clause, n_results=limit ) return[{ "title": meta["title"], "content": doc, "cefr_level": meta["cefr_level"], "difficulty_score": meta["difficulty_score"] } for doc, meta in zip(results["documents"][0], results["metadatas"][0])]

03 阅读理解练习生成

使用Ollama根据文章内容生成阅读理解题目：

python
import ollama import json class ReadingComprehensionGenerator: def __init__(self, model="llama3:8b"): self.model = model def generate_questions(self, article_title, article_content, question_count=5): prompt = f""" 你是一位专业的英语老师。请根据以下文章生成{question_count}道阅读理解题目。 文章标题：{article_title} 文章内容： {article_content} 题目类型包括： 1. 主旨大意题（main idea） 2. 细节理解题（detail） 3. 推理判断题（inference） 4. 词义猜测题（vocabulary） 请以JSON格式输出，不要包含其他内容： {{ "title": "文章标题", "questions": [ {{ "type": "main idea/detail/inference/vocabulary", "question": "题目内容", "options": ["选项A", "选项B", "选项C", "选项D"], "answer": "正确答案（A/B/C/D）", "explanation": "答案解析" }} ] }} """ response = ollama.chat( model=self.model, messages=[{ "role": "user", "content": prompt }] ) try: return json.loads(response["message"]["content"]) except json.JSONDecodeError: returnNone def generate_summary(self, article_content): prompt = f""" 请为以下英语文章生成中文摘要（100-150字），并列出5个重点单词： 文章内容： {article_content} 请以JSON格式输出： {{ "summary": "中文摘要", "key_words": ["单词1", "单词2", "单词3", "单词4", "单词5"] }} """ response = ollama.chat( model=self.model, messages=[{ "role": "user", "content": prompt }] ) try: return json.loads(response["message"]["content"]) except json.JSONDecodeError: returnNone

04 自适应难度调整

根据学习者表现动态调整文章难度：

python
import json import os class ReadingLevelManager: def __init__(self, file_path="reading_level.json"): self.file_path = file_path self.level = "B1" self.accuracy_history = [] self._load() self.cefr_order = ["A1", "A2", "B1", "B2", "C1", "C2"] def _load(self): if os.path.exists(self.file_path): with open(self.file_path, 'r', encoding='utf-8') as f: data = json.load(f) self.level = data.get("level", "B1") self.accuracy_history = data.get("accuracy_history", []) def _save(self): data = { "level": self.level, "accuracy_history": self.accuracy_history } with open(self.file_path, 'w', encoding='utf-8') as f: json.dump(data, f, ensure_ascii=False, indent=2) def record_score(self, correct_count, total_count): accuracy = correct_count / total_count if total_count > 0 else0 self.accuracy_history.append(accuracy) if len(self.accuracy_history) > 10: self.accuracy_history = self.accuracy_history[-10:] self._adjust_level() self._save() def _adjust_level(self): if len(self.accuracy_history) < 5: return avg_accuracy = sum(self.accuracy_history[-5:]) / 5 current_idx = self.cefr_order.index(self.level) if avg_accuracy >= 0.85 and current_idx < len(self.cefr_order)-1: self.level = self.cefr_order[current_idx + 1] print(f"🎉 升级到 {self.level} 级别！") elif avg_accuracy < 0.5 and current_idx > 0: self.level = self.cefr_order[current_idx - 1] print(f"📉 降级到 {self.level} 级别") def get_current_level(self): return{ "level": self.level, "recent_accuracy": round(sum(self.accuracy_history[-5:]) / max(len(self.accuracy_history[-5:]), 1), 2) }

05 阅读学习流程

python
class ReadingComprehensionLearner: def __init__(self): self.library = ArticleLibrary() self.generator = ReadingComprehensionGenerator() self.level_manager = ReadingLevelManager() self.current_article = None self.current_questions = None def get_recommended_article(self): current_level = self.level_manager.get_current_level() articles = self.library.get_articles_by_difficulty(current_level["level"], 1) if articles: self.current_article = articles[0] summary = self.generator.generate_summary(self.current_article["content"]) return{ "article": self.current_article, "summary": summary, "current_level": current_level } else: return{"error": "该级别暂无文章"} def start_quiz(self, question_count=5): if not self.current_article: return{"error": "请先获取文章"} questions = self.generator.generate_questions( self.current_article["title"], self.current_article["content"], question_count ) self.current_questions = questions return questions def submit_answers(self, answers): if not self.current_questions: return{"error": "请先开始测验"} correct_count = 0 results = [] for i, question in enumerate(self.current_questions["questions"]): user_answer = answers.get(str(i), "") is_correct = user_answer.strip().upper() == question["answer"].strip().upper() if is_correct: correct_count += 1 results.append({ "question": question["question"], "your_answer": user_answer, "correct_answer": question["answer"], "correct": is_correct, "explanation": question["explanation"] }) total_count = len(self.current_questions["questions"]) self.level_manager.record_score(correct_count, total_count) return{ "correct_count": correct_count, "total_count": total_count, "accuracy": round(correct_count / total_count, 2), "results": results, "level_change": self.level_manager.get_current_level() }

⚠️ 常见问题

1. 文章难度评估不准确：可以调整DifficultyEvaluator中的CEFR映射规则

2. 题目生成质量不稳定：优化prompt或使用更强大的模型

3. 自适应调整过于频繁：增加_adjust_level中的样本数量要求

4. 文章库为空：使用ArticleLibrary.add_article添加示例文章
