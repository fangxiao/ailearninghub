内容生产团队

这一篇，我们用一个实战案例把多Agent协作串起来：做一个内容生产团队，包括策划、写作、编辑、发布四个Agent。

💡 核心观点：多Agent系统的威力在于专业分工。每个Agent专注自己的领域，通过协作完成复杂任务。

1团队架构

pythonclass ContentProductionTeam:
    def __init__(self):
        self.message_bus = MessageBus()
        self.orchestrator = TeamOrchestrator()

        # 创建各个Agent
        self.planner = PlannerAgent()
        self.writer = WriterAgent()
        self.editor = EditorAgent()
        self.publisher = PublisherAgent()

        # 注册团队成员
        self.orchestrator.add_member(self.planner, AgentRole.PLANNER, ["planning", "research"])
        self.orchestrator.add_member(self.writer, AgentRole.EXECUTOR, ["writing"])
        self.orchestrator.add_member(self.editor, AgentRole.REVIEWER, ["editing", "quality"])
        self.orchestrator.add_member(self.publisher, AgentRole.COORDINATOR, ["publishing"])

    def produce_content(self, topic):
        # 第一步：策划
        plan = self.orchestrator.assign_task(
            {"id": "1", "task": f"策划一篇关于{topic}的文章", "capabilities": ["planning"]},
            self.planner_member
        )

        # 第二步：写作
        draft = self.orchestrator.assign_task(
            {"id": "2", "task": "根据策划写文章草稿", "capabilities": ["writing"]},
            self.writer_member
        )

        # 第三步：编辑
        edited = self.orchestrator.assign_task(
            {"id": "3", "task": "编辑审核文章", "capabilities": ["editing"]},
            self.editor_member
        )

        # 第四步：发布
        result = self.orchestrator.assign_task(
            {"id": "4", "task": "发布文章到平台", "capabilities": ["publishing"]},
            self.publisher_member
        )

        return result

2Agent实现

pythonclass PlannerAgent:
    def execute(self, task):
        prompt = f"""为文章制定写作计划：

任务：{task['task']}

请制定：
1. 文章主题和角度
2. 核心观点（3-5个）
3. 文章结构大纲
4. 需要的研究资料"""

        response = llm.generate(prompt)

class WriterAgent:
    def execute(self, task):
        plan = task.get("plan")
        prompt = f"根据以下计划撰写文章：\n\n{plan}"
        draft = llm.generate(prompt)
        return {"draft": draft, "status": "completed"}

class EditorAgent:
    def execute(self, task):
        draft = task.get("draft")
        prompt = f"""编辑审核以下文章：

{draft}

请检查并修正：
1. 语法错误
2. 逻辑问题
3. 表达不清的地方
4. 格式问题"""

        edited = llm.generate(prompt)
        return {"edited": edited, "status": "completed"}

class PublisherAgent:
    def execute(self, task):
        article = task.get("edited")
        platform = self.select_platform(article)
        result = self.publish(article, platform)
        return {"published": result, "platform": platform}

性能优化

代码跑通了，接下来要考虑性能。Agent系统常见的性能问题有哪些？怎么优化？

💡 核心观点：性能优化三个方向：减少等待、并行处理、合理缓存。抓住这三点，Agent系统效率翻倍。

1并发处理

pythonimport asyncio
from concurrent.futures import ThreadPoolExecutor

class AsyncExecutionEngine:
    def __init__(self, max_workers=10):
        self.executor = ThreadPoolExecutor(max_workers=max_workers)

    async def execute_parallel(self, tasks):
        loop = asyncio.get_event_loop()
        futures = [
            loop.run_in_executor(self.executor, task)
            for task in tasks
        ]
        results = await asyncio.gather(*futures)
        return results

    async def execute_batch(self, batch):
        # 同一批次的任务可以并行执行
        return await self.execute_parallel(batch)

2缓存策略

pythonfrom functools import lru_cache
import hashlib

class ResultCache:
    def __init__(self, max_size=1000):
        self.cache = {}
        self.max_size = max_size

    def _make_key(self, task):
        # 基于任务内容生成缓存key
        content = str(task.get("task")) + str(task.get("context"))
        return hashlib.md5(content.encode()).hexdigest()

    def get(self, task):
        key = self._make_key(task)
        return self.cache.get(key)

    def set(self, task, result):
        if len(self.cache) >= self.max_size:
            # LRU淘汰
            self.cache.pop(next(iter(self.cache)))
        key = self._make_key(task)
        self.cache[key] = result

3流式输出

pythonasync def stream_generate(prompt):
    # 使用流式API减少等待时间
    stream = await openai.ChatCompletion.acreate(
        model="gpt-4",
        messages=[{"role": "user", "content": prompt}],
        stream=True
    )

    accumulated = ""
    async for chunk in stream:
        delta = chunk.choices[0].delta.content
        if delta:
            accumulated += delta
            yield delta  # 实时yield，每个token

💡 性能优化 Checklist：1) 能并行的任务不串行 2) 重复计算要缓存 3) 流式输出减少等待 4) 合理设置超时
