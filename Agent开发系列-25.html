可观测性

Agent系统上线后，怎么知道它运行得好不好？有没有出问题？可观测性就是回答这些问题的。

💡 核心观点：可观测性三板斧：日志、指标、追踪。做好这三点，生产环境的问题无处遁形。

1日志设计

pythonimport logging
import json
from datetime import datetime

class AgentLogger:
    def __init__(self, agent_id):
        self.agent_id = agent_id
        self.logger = logging.getLogger(f"agent.{agent_id}")

    def log(self, level, event, data=None):
        log_entry = {
            "timestamp": datetime.now().isoformat(),
            "agent_id": self.agent_id,
            "event": event,
            "data": data or {}
        }
        if level == "error":
            self.logger.error(json.dumps(log_entry))
        else:
            self.logger.info(json.dumps(log_entry))

    def log_task_start(self, task):
        self.log("info", "task_start", {"task": task})

    def log_task_complete(self, task_id, result):
        self.log("info", "task_complete")">{task_id, result}

    def log_error(self, error, context):
        self.log("error", "error", {"error": str(error), "context": context})
